[rules]
# 规则1：禁止执行任何包含 --force 的 git 命令
[[rules.entries]]
description = "禁止 git push --force"
pattern = "git push.*--force"
action = "forbid"        # 直接拦截，不给你机会
scope = "manual"

# 规则2：执行数据库迁移前先提示确认
[[rules.entries]]
description = "数据库迁移需二次确认"
pattern = "db:migrate|prisma migrate|alembic upgrade"
action = "prompt"        # 弹确认框，你手动批准后执行
scope = "auto"

# 规则3：禁止在 /tmp 或临时目录中写入 .env 文件
[[rules.entries]]
description = "禁止在临时目录写环境变量文件"
pattern = "/tmp/.*\\.env"
action = "forbid"
scope = "auto"
